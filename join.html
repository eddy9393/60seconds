  <script>

    const trackFileInput = document.getElementById("trackFile");

    const audioPreview = document.getElementById("audioPreview");

    const startTimeInput = document.getElementById("startTime");

    const endTimeInput = document.getElementById("endTime");

    const audioHint = document.getElementById("audioHint");

    const joinForm = document.getElementById("joinForm");

    const formStatus = document.getElementById("formStatus");


    let uploadedAudioDuration = 0;

    let uploadedAudioUrl = null;


    trackFileInput.addEventListener("change", (event) => {

      const file = event.target.files[0];

      if (!file) return;


      if (uploadedAudioUrl) {

        URL.revokeObjectURL(uploadedAudioUrl);

      }


      uploadedAudioUrl = URL.createObjectURL(file);

      audioPreview.src = uploadedAudioUrl;

      audioPreview.load();


      audioPreview.onloadedmetadata = () => {

        uploadedAudioDuration = Math.floor(audioPreview.duration || 0);


        startTimeInput.max = Math.max(uploadedAudioDuration - 60, 0);

        startTimeInput.value = 0;

        endTimeInput.value = Math.min(60, uploadedAudioDuration);


        audioHint.textContent =

          uploadedAudioDuration >= 60

            ? `Track loaded. Total length: ${uploadedAudioDuration} seconds. Choose exactly 60 seconds.`

            : `This track is shorter than 60 seconds (${uploadedAudioDuration} seconds).`;

      };

    });


    startTimeInput.addEventListener("input", () => {

      const start = Number(startTimeInput.value || 0);

      endTimeInput.value = start + 60;

    });


    endTimeInput.addEventListener("input", () => {

      const end = Number(endTimeInput.value || 60);

      startTimeInput.value = Math.max(0, end - 60);

    });


    joinForm.addEventListener("submit", (event) => {

      event.preventDefault();


      const artistName = document.getElementById("artistName").value.trim();

      const trackTitle = document.getElementById("trackTitle").value.trim();

      const start = Number(startTimeInput.value || 0);

      const end = Number(endTimeInput.value || 0);


      if (!artistName || !trackTitle) {

        formStatus.className = "status error";

        formStatus.textContent = "Please fill in your artist name and track title.";

        return;

      }


      if (!trackFileInput.files[0]) {

        formStatus.className = "status error";

        formStatus.textContent = "Please upload a track.";

        return;

      }


      if (end - start !== 60) {

        formStatus.className = "status error";

        formStatus.textContent = "Your selected radio section must be exactly 60 seconds.";

        return;

      }


      formStatus.className = "status success";

      formStatus.textContent =

        "Demo complete: your track details are ready. Next step is connecting this page to real storage and submissions.";

    });

  </script>
