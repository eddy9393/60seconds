  <script>

    const supabaseClient = window.supabase.createClient(

      "https://rgoutegbcpjytplqcwze.supabase.co",

      "sb_publishable_255qyDKS77nMU0pbedfa_A_3hdgtEHh"

    );


    const statusEl = document.getElementById("status");

    const debugEl = document.getElementById("debug");

    const trackListEl = document.getElementById("trackList");


    function setStatus(msg, isError = false) {

      statusEl.textContent = msg;

      statusEl.style.color = isError ? "#ff8a8a" : "#ddd";

      console.log(msg);

    }


    function setDebug(msg) {

      debugEl.textContent = msg;

      console.log("DEBUG:", msg);

    }


    async function loadPendingTracks() {

      setDebug("Loading pending tracks from Supabase...");


      const { data, error } = await supabaseClient

        .from("tracks")

        .select("id, title, artist, file_url, status, created_at")

        .eq("status", "pending")

        .order("created_at", { ascending: false });


      if (error) {

        setStatus("Error loading tracks: " + error.message, true);

        return;

      }


      renderTracks(data || []);

    }


    function renderTracks(tracks) {

      trackListEl.innerHTML = "";


      if (!tracks.length) {

        trackListEl.innerHTML = `<div class="empty">No pending tracks right now.</div>`;

        setStatus("No pending tracks found.");

        return;

      }


      setStatus(`${tracks.length} pending track(s) found.`);


      tracks.forEach(track => {

        const div = document.createElement("div");

        div.className = "track";

        div.innerHTML = `

          <h3>${escapeHtml(track.title)}</h3>

          <p><strong>Artist:</strong> ${escapeHtml(track.artist)}</p>

          <p><strong>Status:</strong> ${escapeHtml(track.status)}</p>

          <p><strong>ID:</strong> ${escapeHtml(track.id)}</p>

          <audio controls src="${track.file_url}"></audio>

          <div>

            <button class="gold" data-id="${track.id}" data-action="approve">Approve</button>

          </div>

        `;

        trackListEl.appendChild(div);

      });


      wireButtons();

    }


    function wireButtons() {

      const buttons = document.querySelectorAll("button[data-action='approve']");


      buttons.forEach(button => {

        button.addEventListener("click", async () => {

          const id = button.getAttribute("data-id");

          await approveTrack(id);

        });

      });

    }


    async function approveTrack(trackId) {

      setStatus("Approving track...");

      setDebug("Updating status to approved for track " + trackId);


      const { error } = await supabaseClient

        .from("tracks")

        .update({ status: "approved" })

        .eq("id", trackId);


      if (error) {

        setStatus("Approve failed: " + error.message, true);

        return;

      }


      setStatus("Track approved ✅");

      await loadPendingTracks();

    }


    function escapeHtml(str) {

      return String(str)

        .replaceAll("&", "&amp;")

        .replaceAll("<", "&lt;")

        .replaceAll(">", "&gt;")

        .replaceAll('"', "&quot;")

        .replaceAll("'", "&#039;");

    }
